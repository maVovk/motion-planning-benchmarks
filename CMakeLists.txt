cmake_minimum_required(VERSION 3.22)
project(coal_benchmark LANGUAGES CXX)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
find_package(coal 3 REQUIRED)
add_executable(coal_benchmark main.cpp)
target_compile_features(coal_benchmark PRIVATE cxx_std_17)
target_link_libraries(coal_benchmark PRIVATE coal::coal)
target_compile_options(coal_benchmark PRIVATE -Wall -Wextra -Wpedantic)
